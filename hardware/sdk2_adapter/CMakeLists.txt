cmake_minimum_required(VERSION 3.16)
project(go2_sdk2_adapter LANGUAGES CXX)
find_package(unitree_sdk2 REQUIRED CONFIG)
find_package(Eigen3 REQUIRED CONFIG)
find_package(Boost REQUIRED)
add_library(go2_sdk2_adapter STATIC command_adapter.cpp)
target_compile_features(go2_sdk2_adapter PUBLIC cxx_std_17)
target_include_directories(go2_sdk2_adapter PUBLIC
  include ../../ros2/go2_control_cpp/include)
target_link_libraries(go2_sdk2_adapter PUBLIC unitree_sdk2 Eigen3::Eigen)
target_compile_options(go2_sdk2_adapter PRIVATE -Wall -Wextra -Wpedantic)
add_executable(go2_sdk2_adapter_check command_adapter_check.cpp)
target_include_directories(go2_sdk2_adapter_check PRIVATE ${Boost_INCLUDE_DIRS})
target_link_libraries(go2_sdk2_adapter_check PRIVATE go2_sdk2_adapter)
target_compile_options(go2_sdk2_adapter_check PRIVATE -Wall -Wextra -Wpedantic)
